\begin{frame}[fragile]
    \frametitle{The "final" Workflow}
    Please copy the \altverb{final} workflow from the \altverb{<++course.pathtosolutions++>} directory. (Or copy the directory and the \altverb{data} directory into it.)
    \pause
    \begin{warning}
        A real workflow on a HPC cluster will
        \begin{itemize}
            \item have its code in the \altverb{HOME} directory (or a shared development directory)
            \item and the data in a project, scratch or workspace -- which is a \emph{different} file system
        \end{itemize}
        Then the workflow will be started with the \altverb{--directory} flag indicating the analysis directory.
    \end{warning}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]
    \fametitle{\HandsOn{Start the Workflow}}
    In your \altverb{final} directory with the \altverb{data}, \altverb{envs}, \altverb{profile}, \altverb{rules} and \altverb{scripts} directory:
    \begin{lstlisting}[language=Bash, style=Shell]
$ snakemake -j unlimited \
> --configfile config/config.yaml 
> --sdm conda \
> --workflow-profile profile \ 
> --executor slurm # not necessary with
                   # a global profile
    \end{lstlisting}
\end{frame}
