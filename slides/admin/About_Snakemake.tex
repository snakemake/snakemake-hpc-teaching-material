\subsection{Background}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}
	\frametitle{HPC relevant Features of \Snakemake}
	Apart from the vast and growing number of curated workflows \\
	\vfill
	\begin{columns}[t]
		\begin{column}{0.5\textwidth}
			{\bf General:}
			\begin{itemize}[<+->]
				\item caching of files to re-use (e.\,g. reference downloads)
				\item internal support for scripts (R, Python, shell dialects, $\ldots$)
				\item possible to update from a queue (e.\,g. experimental data live processing)
				\item I/O conservative metadata caching with SQLite
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\pause{\bf Cluster specifics:}
			\begin{itemize}[<+->]
				\item automatic job submission and resource allocation
				\item job submission of SMP, MPI or GPU tasks
				\item minimal strain on \texttt{slurmdbd} or \texttt{slurmctld} for job status checks
				\item SLURM: automated stage-in upon file access pattern annotation
				\item array job support - planned pooling of SMP jobs in (multi) node jobs
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]
	\frametitle{How does \Snakemake on a Cluster work?}
	\begin{columns}[T]
		\begin{column}{.5\textwidth}
			\only<3->{\includegraphics[width=0.95\textwidth]{misc/cluster_scheme.png}}
		\end{column}
		\begin{column}{.5\textwidth}
			\begin{itemize}[<+->]
				\item Snakemake is triggered on the command line:
				\begin{lstlisting}[language=Bash, style=Shell]
$ snakemake [<arguments>]
				\end{lstlisting} 
			    \item users need to fill in the parameters of their workflow (e.\,g. input path(s) in a file)
			    \item Snakemake will run on the login-node and spawn your jobs on the cluster using a
			    \item cluster specific configuration file 
			\end{itemize}
		    \pause
			It is capable to remove temporary files and support various archiving systems.
		\end{column}
	\end{columns}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]
	\frametitle{Global configuration}
	\Snakemake listens to a variable \altverb{SNAKEMAKE_PROFILE}. This may point to 
	\begin{itemize}
		\item \altverb{/etc/xdg/snakemake} (globally on a login node)
		\item or \altverb{$HOME/.config/snakemake}
	\end{itemize}
	\begin{hint}
       This will not interfere with other workflow engines!
	\end{hint}
\end{frame}
